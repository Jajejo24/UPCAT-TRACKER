                const g = Object.values(grades).map(Number);

                return avg;
            }, [grades]);

            // Convert HS Grade percentage to UP Grade scale (approx 90-100% maps to 1.0 - 2.0 scale)
            // standard formula approximation: UPG Grade = 1.0 + (100 - Grade) * 0.1
            const hsGwaInUPScale = useMemo(() => {
                const upScale = 1.0 + (100 - hsGwaAverage) * 0.08;
                return Math.max(1.0, Math.min(3.5, upScale));
            }, [hsGwaAverage]);

            const mockInUPScale = useMemo(() => {
                const upScale = 1.0 + (100 - mockCompositePercent) * 0.075;

            }, [mockCompositePercent]);

            const estimatedUPG = useMemo(() => {
                const result = (0.40 * hsGwaInUPScale) + (0.60 * mockInUPScale);

        // ==========================================
        // 2. INTERACTIVE MOCK EXAM SIMULATOR
        // ==========================================
        function MockExamSimulator({ questions, setQuestions, setMockScores, setExamLogs }) {
            const [testState, setTestState] = useState('idle'); // 'idle', 'running', 'finished'
            const [currentIndex, setCurrentIndex] = useState(0);
            const [userAnswers, setUserAnswers] = useState({});
            const [flagged, setFlagged] = useState({});
            const [timeLeft, setTimeLeft] = useState(600); // 10 minutes in seconds
            const [isPaused, setIsPaused] = useState(false);
            const [importError, setImportError] = useState('');

            const timerRef = useRef(null);

            // Start Mock Exam
            const startTest = () => {
                setUserAnswers({});
                setFlagged({});
                setCurrentIndex(0);
                setTimeLeft(questions.length * 60); // 1 min per question default
                setIsPaused(false);
                setTestState('running');
            };

            // Timer Loop
            useEffect(() => {
                if (testState === 'running' && !isPaused) {
                    timerRef.current = setInterval(() => {
                        setTimeLeft(prev => {
                            if (prev <= 1) {
                                clearInterval(timerRef.current);
                                finishTest();
                                return 0;
                            }
                            return prev - 1;
                        });
                    }, 1000);
                }
                return () => clearInterval(timerRef.current);
            }, [testState, isPaused]);

            // Select answer
            const handleSelectAnswer = (optionIdx) => {
                setUserAnswers(prev => ({
                    ...prev,
                    [currentIndex]: optionIdx
                }));
            };

            // Toggle Flag
            const toggleFlag = (idx) => {
                setFlagged(prev => ({
                    ...prev,
                    [idx]: !prev[idx]
                }));
            };

            // Finish Test & Calculate Score Breakdown
            const finishTest = () => {
                setTestState('finished');
                
                // Calculate scores per subject
                let totalCorrect = 0;
                const subjectStats = {};

                questions.forEach((q, idx) => {
                    const isCorrect = userAnswers[idx] === q.correctAnswer;
                    if (isCorrect) totalCorrect++;

                    if (!subjectStats[q.subject]) {
                        subjectStats[q.subject] = { correct: 0, total: 0 };
                    }
                    subjectStats[q.subject].total++;
                    if (isCorrect) subjectStats[q.subject].correct++;
                });

                // Update global mock scores percentage
                setMockScores(prev => {
                    const updated = { ...prev };
                    Object.keys(subjectStats).forEach(sub => {
                        const pct = Math.round((subjectStats[sub].correct / subjectStats[sub].total) * 100);
                        if (updated[sub] !== undefined) {
                            updated[sub] = pct;
                        }
                    });
                    return updated;
                });

                // Save to history log
                const newLog = {
                    title: `Mock Exam (${questions.length} Qs)`,
                    date: new Date().toLocaleDateString() + ' ' + new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
                    score: totalCorrect,
                    total: questions.length
                };
                setExamLogs(prev => [...prev, newLog]);
            };

            // JSON File Import Handler
            const handleFileUpload = (e) => {
                const file = e.target.files[0];
                if (!file) return;

                const reader = new FileReader();
                reader.onload = (event) => {
                    try {
                        const importedData = JSON.parse(event.target.result);
                        if (Array.isArray(importedData) && importedData.length > 0 && importedData[0].question && importedData[0].options) {
                            setQuestions(importedData);
                            setImportError('');
                            alert(`Successfully loaded ${importedData.length} custom questions!`);
                        } else {
                            setImportError('Invalid JSON structure. Must be an array of questions with keys: id, question, options, correctAnswer, subject, explanation.');
                        }
                    } catch (err) {
                        setImportError('Failed to parse JSON file. Please check syntax.');
                    }
                };
                reader.readAsText(file);
            };

            // Format Timer Text mm:ss
            const formatTime = (secs) => {
                const m = Math.floor(secs / 60);
                const s = secs % 60;
                return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
            };

            const currentQ = questions[currentIndex];

            return (
                <div className="space-y-6">
                    {/* IDLE / START OVERVIEW */}
                    {testState === 'idle' && (
                        <div className="glass-card rounded-2xl p-8 max-w-2xl mx-auto text-center space-y-6">
                            <div className="w-16 h-16 rounded-2xl bg-up-maroon/10 text-up-maroon dark:text-up-gold flex items-center justify-center mx-auto text-2xl font-bold">
                                📝
                            </div>
                            <div>
                                <h2 className="text-xl font-bold text-gray-900 dark:text-white">UPCAT Mock Exam Simulator</h2>
                                <p className="text-xs text-gray-500 dark:text-gray-400 mt-1 max-w-md mx-auto">
                                    Simulate real exam pressure with timed questions, question bookmarking, auto score tracking, and instant detailed solutions.
                                </p>
                            </div>

                            <div className="p-4 bg-gray-50 dark:bg-zinc-900/60 rounded-xl text-left text-xs space-y-2 border border-gray-200 dark:border-zinc-800">
                                <div className="flex justify-between">
                                    <span className="text-gray-500">Total Questions Loaded:</span>
                                    <strong className="font-semibold">{questions.length} Questions</strong>
                                </div>
                                <div className="flex justify-between">
                                    <span className="text-gray-500">Estimated Duration:</span>
                                    <strong className="font-semibold">{Math.ceil(questions.length * 1)} Minutes</strong>
                                </div>
                                <div className="flex justify-between">
                                    <span className="text-gray-500">Subjects Covered:</span>
                                    <strong className="font-semibold text-up-maroon dark:text-up-gold">Math, Science, Read Comp, Lang Prof</strong>
                                </div>
                            </div>

                            {/* CUSTOM FILE IMPORT SECTION */}
                            <div className="pt-2 border-t border-gray-200 dark:border-zinc-800 text-left">
                                <label className="block text-xs font-semibold text-gray-700 dark:text-gray-300 mb-2">
                                    Import Custom JSON Question Bank:
                                </label>
                                <input 
                                    type="file" 
                                    accept=".json"
                                    onChange={handleFileUpload}
                                    className="block w-full text-xs text-gray-500 file:mr-4 file:py-2 file:px-4 file:rounded-lg file:border-0 file:text-xs file:font-semibold file:bg-up-maroon file:text-white hover:file:bg-up-maroonDark cursor-pointer"
                                />
                                {importError && <p className="text-rose-500 text-[11px] mt-2">{importError}</p>}
                            </div>

                            <button
                                onClick={startTest}
                                className="w-full py-3 bg-up-maroon hover:bg-up-maroonDark text-white font-bold rounded-xl shadow-lg transition-transform active:scale-95 text-sm"
                            >
                                Start Timed Mock Exam
                            </button>
                        </div>
                    )}

                    {/* RUNNING EXAM INTERFACE */}
                    {testState === 'running' && currentQ && (
                        <div className="grid grid-cols-1 lg:grid-cols-4 gap-6">
                            {/* MAIN QUESTION DISPLAY */}
                            <div className="lg:col-span-3 space-y-4">
                                {/* TOOLBAR TIMER & PROGRESS */}
                                <div className="glass-card rounded-2xl p-4 flex items-center justify-between shadow-sm">
                                    <div className="flex items-center space-x-3">
                                        <span className="text-xs px-2.5 py-1 rounded-full bg-up-maroon/10 text-up-maroon dark:bg-up-gold/20 dark:text-up-gold font-bold">
                                            {currentQ.subject}
                                        </span>
                                        <span className="text-xs text-gray-400 font-medium">Topic: {currentQ.topic || 'General'}</span>
                                    </div>

                                    {/* TIMER */}
                                    <div className="flex items-center space-x-3">
                                        <button 
                                            onClick={() => setIsPaused(!isPaused)}
                                            className="px-2.5 py-1 text-xs font-semibold rounded-lg bg-gray-200 dark:bg-zinc-800 hover:bg-gray-300 transition"
                                        >
                                            {isPaused ? 'Resume' : 'Pause'}
                                        </button>
                                        <div className="flex items-center space-x-1.5 px-3 py-1 rounded-lg bg-up-maroon text-white font-mono font-bold text-sm">
                                            <i data-lucide="clock" className="w-4 h-4"></i>
                                            <span>{formatTime(timeLeft)}</span>
                                        </div>
                                    </div>
                                </div>

                                {/* QUESTION CARD */}
                                <div className="glass-card rounded-2xl p-6 shadow-sm space-y-6">

                                            </button>

                            <div className="glass-card rounded-2xl p-4 shadow-sm h-fit space-y-4">

                                        const isFlagged = flagged[idx];

                                        return (
                                            <button
                                                key={idx}
                                                onClick={() => setCurrentIndex(idx)}
                                                className={`h-9 w-full rounded-lg text-xs font-bold transition-all relative ${
                                                    isCurrent
                                                        ? 'ring-2 ring-up-maroon dark:ring-up-gold bg-up-maroon text-white'
                                                        : isAnswered

                                                {isFlagged && (

                                    <div className="flex items-center space-x-2">

                                        <span className="w-3 h-3 bg-amber-500 rounded-full"></span>

        // 3. SYLLABUS ROADMAP & TOPIC CHECKLIST
        // ==========================================
        function SyllabusTab({ completedTopics, setCompletedTopics }) {
            const toggleTopic = (topicId) => {
                setCompletedTopics(prev => ({
                    ...prev,
                    [topicId]: !prev[topicId]
                }));
            };

            return (
                <div className="space-y-6">
                    <div className="flex flex-col md:flex-row items-start md:items-center justify-between gap-4">
                        <div>
                            <h2 className="text-xl font-bold text-gray-900 dark:text-white">UPCAT Core Syllabus Checklist</h2>
                            <p className="text-xs text-gray-500 dark:text-gray-400 mt-1">
                                High-yield topics curated by Brain Train methodologies. Track your mastery per subtest.
                            </p>
                        </div>
                    </div>

                    {/* SUBJECT CARDS GRID */}
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
                        {Object.keys(SYLLABUS_DATA).map(subject => {
                            const topics = SYLLABUS_DATA[subject];
                            const completedCount = topics.filter(t => completedTopics[t.id]).length;
                            const progressPct = Math.round((completedCount / topics.length) * 100);

                            return (
                                <div key={subject} className="glass-card rounded-2xl p-6 shadow-sm space-y-4">
                                    <div className="flex justify-between items-center">
                                        <h3 className="text-base font-bold text-gray-900 dark:text-white flex items-center gap-2">
                                            <i data-lucide="book-open" className="w-4 h-4 text-up-maroon dark:text-up-gold"></i>
                                            {subject}
                                        </h3>
                                        <span className="text-xs font-bold font-mono text-up-maroon dark:text-up-gold">
                                            {completedCount} / {topics.length} ({progressPct}%)
                                        </span>
                                    </div>

                                    {/* PROGRESS BAR */}
                                    <div className="w-full bg-gray-200 dark:bg-zinc-800 rounded-full h-2 overflow-hidden">
                                        <div 
                                            className="bg-up-maroon dark:bg-up-gold h-full transition-all duration-300"
                                            style={{ width: `${progressPct}%` }}
                                        ></div>
                                    </div>

                                    {/* CHECKLIST TOPICS */}
                                    <div className="space-y-2 pt-2">
                                        {topics.map(topic => {
                                            const isChecked = !!completedTopics[topic.id];
                                            return (
                                                <label 
                                                    key={topic.id}
                                                    className={`flex items-center justify-between p-3 rounded-xl border transition-all cursor-pointer ${
                                                        isChecked 
                                                            ? 'bg-up-green/5 border-up-green/30 text-gray-900 dark:text-white' 
                                                            : 'bg-white dark:bg-zinc-900 border-gray-200 dark:border-zinc-800 text-gray-700 dark:text-gray-300 hover:border-gray-300'
                                                    }`}
                                                >
                                                    <div className="flex items-center space-x-3">
                                                        <input 
                                                            type="checkbox" 
                                                            checked={isChecked}
                                                            onChange={() => toggleTopic(topic.id)}
                                                            className="w-4 h-4 text-up-maroon rounded focus:ring-up-maroon"
                                                        />
                                                        <span className={`text-xs font-medium ${isChecked ? 'line-through opacity-70' : ''}`}>
                                                            {topic.title}
                                                        </span>
                                                    </div>
                                                    <span className="text-[10px] px-2 py-0.5 rounded-md bg-gray-100 dark:bg-zinc-800 text-gray-500 font-semibold">
                                                        {topic.sub}
                                                    </span>
                                                </label>
                                            );
                                        })}
                                    </div>
                                </div>
                            );
                        })}
                    </div>
                </div>
            );
        }

        // ==========================================
        // 4. EMBEDDED POMODORO TIMER & TRACKER
        // ==========================================
        function PomodoroTab({ studyMinutes, setStudyMinutes, setStreak }) {
            const [minutes, setMinutes] = useState(25);
            const [seconds, setSeconds] = useState(0);
            const [isActive, setIsActive] = useState(false);
            const [mode, setMode] = useState('study'); // 'study' (25m) or 'break' (5m)

            const timerRef = useRef(null);

            // Web Audio Chime Sound Synthesizer
            const playChime = () => {
                try {
                    const ctx = new (window.AudioContext || window.webkitAudioContext)();
                    const osc = ctx.createOscillator();
                    const gain = ctx.createGain();
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(587.33, ctx.currentTime); // D5
                    osc.frequency.exponentialRampToValueAtTime(880, ctx.currentTime + 0.3); // A5
                    gain.gain.setValueAtTime(0.3, ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.8);
                    osc.connect(gain);
                    gain.connect(ctx.destination);
                    osc.start();
                    osc.stop(ctx.currentTime + 0.8);
                } catch (e) {
                    console.log('Audio chime unavailable', e);
                }
            };

            useEffect(() => {
                if (isActive) {
                    timerRef.current = setInterval(() => {
                        if (seconds > 0) {
                            setSeconds(seconds - 1);
                        } else if (minutes > 0) {
                            setMinutes(minutes - 1);
                            setSeconds(59);
                        } else {
                            // Completed session
                            playChime();
                            clearInterval(timerRef.current);
                            setIsActive(false);

                            if (mode === 'study') {
                                setStudyMinutes(prev => prev + 25);
                                alert("Great job! 25-minute study session logged.");
                                switchMode('break');
                            } else {
                                alert("Break finished! Ready for the next focused sprint?");
                                switchMode('study');
                            }
                        }
                    }, 1000);
                } else {
                    clearInterval(timerRef.current);
                }
                return () => clearInterval(timerRef.current);
            }, [isActive, minutes, seconds, mode]);

            const switchMode = (newMode) => {
                setMode(newMode);
                setIsActive(false);
                if (newMode === 'study') {
                    setMinutes(25);
                    setSeconds(0);
                } else {
                    setMinutes(5);
                    setSeconds(0);
                }
            };

            const toggleTimer = () => setIsActive(!isActive);

            const resetTimer = () => {
                setIsActive(false);
                setMinutes(mode === 'study' ? 25 : 5);
                setSeconds(0);
            };

            return (
                <div className="max-w-xl mx-auto space-y-6">
                    <div className="glass-card rounded-2xl p-8 text-center shadow-sm space-y-6">
                        <h2 className="text-xl font-bold text-gray-900 dark:text-white">Brain Train Pomodoro Clock</h2>
                        
                        {/* MODE TOGGLES */}
                        <div className="flex justify-center space-x-2">
                            <button 
                                onClick={() => switchMode('study')}
                                className={`px-4 py-2 rounded-xl text-xs font-bold transition ${
                                    mode === 'study' ? 'bg-up-maroon text-white shadow' : 'bg-gray-100 dark:bg-zinc-800 text-gray-600'
                                }`}
                            >
                                📖 25m Focus Sprint
                            </button>
                            <button 
                                onClick={() => switchMode('break')}
                                className={`px-4 py-2 rounded-xl text-xs font-bold transition ${
                                    mode === 'break' ? 'bg-up-green text-white shadow' : 'bg-gray-100 dark:bg-zinc-800 text-gray-600'
                                }`}
                            >
                                ☕ 5m Brain Break
                            </button>
                        </div>

                        {/* BIG COUNTDOWN DISPLAY */}
                        <div className="py-8">
                            <div className="text-6xl font-black font-mono tracking-wider text-gray-900 dark:text-white">
                                {minutes.toString().padStart(2, '0')}:{seconds.toString().padStart(2, '0')}
                            </div>
                            <p className="text-xs text-gray-400 mt-2 font-medium">
                                {mode === 'study' ? 'Keep your focus sharp on UPCAT core subjects' : 'Relax your eyes and hydrate!'}
                            </p>
                        </div>

                        {/* TIMER CONTROLS */}
                        <div className="flex justify-center space-x-4">
                            <button
                                onClick={toggleTimer}
                                className={`px-8 py-3 rounded-xl font-bold text-sm text-white shadow-lg transition-transform active:scale-95 ${
                                    isActive ? 'bg-amber-600 hover:bg-amber-700' : 'bg-up-maroon hover:bg-up-maroonDark'
                                }`}
                            >
                                {isActive ? 'Pause Session' : 'Start Focus Clock'}
                            </button>
                            <button
                                onClick={resetTimer}
                                className="px-5 py-3 rounded-xl border border-gray-300 dark:border-zinc-700 font-semibold text-xs text-gray-600 dark:text-gray-300 hover:bg-gray-100 dark:hover:bg-zinc-800"
                            >
                                Reset
                            </button>
                        </div>
                    </div>

                    {/* TOTAL LOGGED HOURS STAT CARD */}
                    <div className="glass-card rounded-2xl p-6 shadow-sm flex items-center justify-between">
                        <div>
                            <span className="text-xs font-bold text-gray-500 uppercase">Total Logged Focus Time</span>
                            <div className="text-2xl font-black text-up-green dark:text-emerald-400">
                                {(studyMinutes / 60).toFixed(1)} Hours
                            </div>
                        </div>
                        <div className="p-3 bg-up-green/10 text-up-green rounded-xl">
                            <i data-lucide="award" className="w-6 h-6"></i>
                        </div>
                    </div>
                </div>
            );
        }
